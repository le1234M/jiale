useEffect(() => {
  try {
    const saved = localStorage.getItem('jiaolo_copy_history')
    if (saved) setHistory(JSON.parse(saved))
  } catch(e) {}
}, [])
// 保存记录函数
const saveToHistory = (data) => {
  const record = {
    id: Date.now(),
    time: new Date().toLocaleString('zh-CN', { month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit' }),
    storeName: data.storeName || '未命名',
    outputTypes: data.outputTypes || [],
    content: data.content || '',
  }
  const newHistory = [record, ...history].slice(0, 10) // 最多保留10条
  setHistory(newHistory)
  localStorage.setItem('jiaolo_copy_history', JSON.stringify(newHistory))
}
// 删除记录
const deleteHistory = (id) => {
  const newHistory = history.filter(h => h.id !== id)
  setHistory(newHistory)
  localStorage.setItem('jiaolo_copy_history', JSON.stringify(newHistory))
}
